AzureDiagnostics
| where ResourceProvider == 'MICROSOFT.KEYVAULT' and (OperationName startswith "SecretNearExpiry" or OperationName startswith "SecretExpired")
| project
    ResourceId,
    SecretName = column_ifexists("eventGridEventProperties_data_ObjectName_s", ''), 
    OperationName,
    ExpiryDate = unixtime_seconds_todatetime(column_ifexists("eventGridEventProperties_data_EXP_d", 0))
